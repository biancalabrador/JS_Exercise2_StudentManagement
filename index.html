import { getStudentsAPI, addStudentsAPI } from './api.js'; let localStudents = []; const studentForm = document.getElementById('studentForm'); const studentTableBody = document.getElementById('studentTableBody'); const renderTable = (dataArray) => { studentTableBody.innerHTML = ''; if (!dataArray || dataArray.length === 0) { const emptyRow = document.createElement('tr'); emptyRow.innerHTML = ` No student records found. `; studentTableBody.appendChild(emptyRow); return; } dataArray.forEach((student, index) => { const row = document.createElement('tr'); row.innerHTML = ` ${index + 1} ${student.firstName} ${student.lastName} ${student.email} `; studentTableBody.appendChild(row); }); }; studentForm.addEventListener('submit', (event) => { event.preventDefault(); const firstNameInput = document.getElementById('firstName'); const lastNameInput = document.getElementById('lastName'); const emailInput = document.getElementById('email'); const studentData = { firstName: firstNameInput.value.trim(), lastName: lastNameInput.value.trim(), email: emailInput.value.trim() }; if (!studentData.firstName || !studentData.lastName || !studentData.email) { alert('Please fill out all input fields!'); return; } const savedStudent = addStudentsAPI(studentData); localStudents.push(savedStudent); renderTable(localStudents); studentForm.reset(); }); const initApp = () => { localStudents = [...getStudentsAPI()]; renderTable(localStudents); }; initApp();
